    61→/*  ANIMATED SECTION WRAPPER                                          */
    62→/* ------------------------------------------------------------------ */
    63→
    64→function FadeInSection({
    65→  children,
    66→  className = "",
    67→  delay = 0,
    68→}: {
    69→  children: React.ReactNode;
    70→  className?: string;
    71→  delay?: number;
    72→}) {
    73→  const ref = useRef<HTMLDivElement>(null);
    74→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    75→  return (
    76→    <motion.div
    77→      ref={ref}
    78→      initial={{ opacity: 0, y: 40 }}
    79→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
    80→      transition={{ duration: 0.6, ease: "easeOut", delay }}
    81→      className={className}
    82→    >
    83→      {children}
    84→    </motion.div>
    85→  );
    86→}

   109→
   110→function Navigation() {
   111→  const { t, toggle, lang } = useLang();
   112→  const [scrolled, setScrolled] = useState(false);
   113→  const [mobileOpen, setMobileOpen] = useState(false);
   114→
   115→  useEffect(() => {
   116→    const onScroll = () => setScrolled(window.scrollY > 40);
   117→    window.addEventListener("scroll", onScroll, { passive: true });
   118→    return () => window.removeEventListener("scroll", onScroll);
   119→  }, []);
   120→
   121→  return (
   122→    <motion.header
   123→      initial={{ y: -80 }}
   124→      animate={{ y: 0 }}
   125→      transition={{ duration: 0.5, ease: "easeOut" }}
   126→      className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
   127→        scrolled
   128→          ? "bg-white/80 backdrop-blur-md shadow-[0_2px_20px_rgba(0,0,0,0.06)]"
   129→          : "bg-transparent"
   130→      }`}
   131→    >
   132→      <nav className="mx-auto max-w-6xl px-6 h-16 flex items-center justify-between">
   133→        <a
   134→          href="#hero"
   135→          className="font-[family-name:var(--font-heading)] font-bold text-lg tracking-tight"
   136→          style={{ color: "var(--pastel-sage)" }}
   137→        >
   138→          {t.navLogo}
   139→        </a>
   140→
   141→        <div className="flex items-center gap-6">
   142→          {/* Desktop nav */}
   143→          <ul className="hidden md:flex items-center gap-8">
   144→            {t.navItems.map((item) => (
   145→              <li key={item.href}>
   146→                <a
   147→                  href={item.href}
   148→                  className="text-sm font-medium text-[var(--muted-foreground)] hover:text-[var(--foreground)] transition-colors duration-200"
   149→                >
   150→                  {item.label}
   151→                </a>
   152→              </li>
   153→            ))}
   154→          </ul>
   155→
   156→          {/* Language toggle */}
   157→          <button
   158→            onClick={toggle}
   159→            className="flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-semibold transition-colors duration-200 hover:bg-[var(--secondary)]"
   160→            style={{ color: "var(--muted-foreground)" }}
   161→            aria-label={t.toggleAriaLabel}
   162→          >
   163→            <Languages size={14} />
   164→            <span>{t.langLabel}</span>
   165→          </button>
   166→
   167→          {/* Mobile toggle */}
   168→          <button
   169→            className="md:hidden p-2 rounded-lg hover:bg-[var(--secondary)] transition"
   170→            onClick={() => setMobileOpen(!mobileOpen)}
   171→            aria-label="Toggle menu"
   172→          >
   173→            {mobileOpen ? <X size={22} /> : <Menu size={22} />}
   174→          </button>
   175→        </div>
   176→      </nav>
   177→
   178→      {/* Mobile menu */}
   179→      <AnimatePresence>
   180→        {mobileOpen && (
   181→          <motion.div
   182→            initial={{ opacity: 0, height: 0 }}
   183→            animate={{ opacity: 1, height: "auto" }}
   184→            exit={{ opacity: 0, height: 0 }}
   185→            className="md:hidden bg-white/95 backdrop-blur-md border-b border-[var(--border)]"
   186→          >
   187→            <ul className="flex flex-col px-6 py-4 gap-3">
   188→              {t.navItems.map((item) => (
   189→                <li key={item.href}>
   190→                  <a
   191→                    href={item.href}
   192→                    onClick={() => setMobileOpen(false)}
   193→                    className="block py-2 text-sm font-medium text-[var(--muted-foreground)] hover:text-[var(--foreground)] transition-colors"
   194→                  >
   195→                    {item.label}
   196→                  </a>
   197→                </li>
   198→              ))}
   199→            </ul>
   200→          </motion.div>
   201→        )}
   202→      </AnimatePresence>
   203→    </motion.header>
   204→  );
   205→}

   694→/* ------------------------------------------------------------------ */
   695→
   696→function ScrollToTop() {
   697→  const { t } = useLang();
   698→  const [visible, setVisible] = useState(false);
   699→  useEffect(() => {
   700→    const onScroll = () => setVisible(window.scrollY > 600);
   701→    window.addEventListener("scroll", onScroll, { passive: true });
   702→    return () => window.removeEventListener("scroll", onScroll);
   703→  }, []);
   704→
   705→  return (
   706→    <AnimatePresence>
   707→      {visible && (
   708→        <motion.button
   709→          initial={{ opacity: 0, scale: 0.8 }}
   710→          animate={{ opacity: 1, scale: 1 }}
   711→          exit={{ opacity: 0, scale: 0.8 }}
   712→          onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
   713→          className="fixed bottom-6 right-6 z-50 w-11 h-11 rounded-full flex items-center justify-center shadow-lg hover:shadow-xl transition-shadow duration-300"
   714→          style={{ background: "var(--pastel-sage)", color: "#fff" }}
   715→          aria-label={t.heroScrollAria}
   716→        >
   717→          <ArrowUp size={18} />
   718→        </motion.button>
   719→      )}
   720→    </AnimatePresence>
   721→  );
   722→}
