   196→/*  ANIMATED SECTION WRAPPER                                          */
   197→/* ------------------------------------------------------------------ */
   198→
   199→function FadeInSection({
   200→  children,
   201→  className = "",
   202→  delay = 0,
   203→}: {
   204→  children: React.ReactNode;
   205→  className?: string;
   206→  delay?: number;
   207→}) {
   208→  const ref = useRef<HTMLDivElement>(null);
   209→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   210→  return (
   211→    <motion.div
   212→      ref={ref}
   213→      initial={{ opacity: 0, y: 40 }}
   214→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   215→      transition={{ duration: 0.6, ease: "easeOut", delay }}
   216→      className={className}
   217→    >
   218→      {children}
   219→    </motion.div>
   220→  );
   221→}

   226→
   227→function Navigation() {
   228→  const [scrolled, setScrolled] = useState(false);
   229→  const [mobileOpen, setMobileOpen] = useState(false);
   230→
   231→  useEffect(() => {
   232→    const onScroll = () => setScrolled(window.scrollY > 40);
   233→    window.addEventListener("scroll", onScroll, { passive: true });
   234→    return () => window.removeEventListener("scroll", onScroll);
   235→  }, []);
   236→
   237→  return (
   238→    <motion.header
   239→      initial={{ y: -80 }}
   240→      animate={{ y: 0 }}
   241→      transition={{ duration: 0.5, ease: "easeOut" }}
   242→      className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
   243→        scrolled
   244→          ? "bg-white/80 backdrop-blur-md shadow-[0_2px_20px_rgba(0,0,0,0.06)]"
   245→          : "bg-transparent"
   246→      }`}
   247→    >
   248→      <nav className="mx-auto max-w-6xl px-6 h-16 flex items-center justify-between">
   249→        <a
   250→          href="#hero"
   251→          className="font-[family-name:var(--font-heading)] font-bold text-lg tracking-tight"
   252→          style={{ color: "var(--pastel-sage)" }}
   253→        >
   254→          Н. Иванов
   255→        </a>
   256→
   257→        {/* Desktop nav */}
   258→        <ul className="hidden md:flex items-center gap-8">
   259→          {NAV_ITEMS.map((item) => (
   260→            <li key={item.href}>
   261→              <a
   262→                href={item.href}
   263→                className="text-sm font-medium text-[var(--muted-foreground)] hover:text-[var(--foreground)] transition-colors duration-200"
   264→              >
   265→                {item.label}
   266→              </a>
   267→            </li>
   268→          ))}
   269→        </ul>
   270→
   271→        {/* Mobile toggle */}
   272→        <button
   273→          className="md:hidden p-2 rounded-lg hover:bg-[var(--secondary)] transition"
   274→          onClick={() => setMobileOpen(!mobileOpen)}
   275→          aria-label="Toggle menu"
   276→        >
   277→          {mobileOpen ? <X size={22} /> : <Menu size={22} />}
   278→        </button>
   279→      </nav>
   280→
   281→      {/* Mobile menu */}
   282→      <AnimatePresence>
   283→        {mobileOpen && (
   284→          <motion.div
   285→            initial={{ opacity: 0, height: 0 }}
   286→            animate={{ opacity: 1, height: "auto" }}
   287→            exit={{ opacity: 0, height: 0 }}
   288→            className="md:hidden bg-white/95 backdrop-blur-md border-b border-[var(--border)]"
   289→          >
   290→            <ul className="flex flex-col px-6 py-4 gap-3">
   291→              {NAV_ITEMS.map((item) => (
   292→                <li key={item.href}>
   293→                  <a
   294→                    href={item.href}
   295→                    onClick={() => setMobileOpen(false)}
   296→                    className="block py-2 text-sm font-medium text-[var(--muted-foreground)] hover:text-[var(--foreground)] transition-colors"
   297→                  >
   298→                    {item.label}
   299→                  </a>
   300→                </li>
   301→              ))}
   302→            </ul>
   303→          </motion.div>
   304→        )}
   305→      </AnimatePresence>
   306→    </motion.header>
   307→  );
   308→}

   945→/*  SCROLL-TO-TOP                                                      */
   946→/* ------------------------------------------------------------------ */
   947→
   948→function ScrollToTop() {
   949→  const [visible, setVisible] = useState(false);
   950→
   951→  useEffect(() => {
   952→    const onScroll = () => setVisible(window.scrollY > 600);
   953→    window.addEventListener("scroll", onScroll, { passive: true });
   954→    return () => window.removeEventListener("scroll", onScroll);
   955→  }, []);
   956→
   957→  return (
   958→    <AnimatePresence>
   959→      {visible && (
   960→        <motion.button
   961→          initial={{ opacity: 0, scale: 0.8 }}
   962→          animate={{ opacity: 1, scale: 1 }}
   963→          exit={{ opacity: 0, scale: 0.8 }}
   964→          onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
   965→          className="fixed bottom-6 right-6 z-50 w-11 h-11 rounded-full flex items-center justify-center shadow-lg hover:shadow-xl transition-shadow duration-300"
   966→          style={{ background: "var(--pastel-sage)", color: "#fff" }}
   967→          aria-label="Наверх"
   968→        >
   969→          <ArrowUp size={18} />
   970→        </motion.button>
   971→      )}
   972→    </AnimatePresence>
   973→  );
   974→}
