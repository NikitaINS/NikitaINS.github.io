   199→   199→function FadeInSection({
   200→   200→  children,
   201→   201→  className = "",
   202→   202→  delay = 0,
   203→   203→}: {
   204→   204→  children: React.ReactNode;
   205→   205→  className?: string;
   206→   206→  delay?: number;
   207→   207→}) {
   208→   208→  const ref = useRef<HTMLDivElement>(null);
   209→   209→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   210→   210→  return (
   211→   211→    <motion.div
   212→   212→      ref={ref}
   213→   213→      initial={{ opacity: 0, y: 40 }}
   214→   214→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   215→   215→      transition={{ duration: 0.6, ease: "easeOut", delay }}
   216→   216→      className={className}
   217→   217→    >
   218→   218→      {children}
   219→   219→    </motion.div>
   220→   220→  );
   221→   221→}

   948→   948→function ScrollToTop() {
   949→   949→  const [visible, setVisible] = useState(false);
   950→   950→
   951→   951→  useEffect(() => {
   952→   952→    const onScroll = () => setVisible(window.scrollY > 600);
   953→   953→    window.addEventListener("scroll", onScroll, { passive: true });
   954→   954→    return () => window.removeEventListener("scroll", onScroll);
   955→   955→  }, []);
   956→   956→
   957→   957→  return (
   958→   958→    <AnimatePresence>
   959→   959→      {visible && (
   960→   960→        <motion.button
   961→   961→          initial={{ opacity: 0, scale: 0.8 }}
   962→   962→          animate={{ opacity: 1, scale: 1 }}
   963→   963→          exit={{ opacity: 0, scale: 0.8 }}
   964→   964→          onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
   965→   965→          className="fixed bottom-6 right-6 z-50 w-11 h-11 rounded-full flex items-center justify-center shadow-lg hover:shadow-xl transition-shadow duration-300"
   966→   966→          style={{ background: "var(--pastel-sage)", color: "#fff" }}
   967→   967→          aria-label="Наверх"
   968→   968→        >
   969→   969→          <ArrowUp size={18} />
   970→   970→        </motion.button>
   971→   971→      )}
   972→   972→    </AnimatePresence>
   973→   973→  );
   974→   974→}
