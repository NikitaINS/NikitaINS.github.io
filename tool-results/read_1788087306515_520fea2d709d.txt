   244→   244→function FadeInSection({
   245→   245→  children,
   246→   246→  className = "",
   247→   247→  delay = 0,
   248→   248→}: {
   249→   249→  children: React.ReactNode;
   250→   250→  className?: string;
   251→   251→  delay?: number;
   252→   252→}) {
   253→   253→  const ref = useRef<HTMLDivElement>(null);
   254→   254→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   255→   255→  return (
   256→   256→    <motion.div
   257→   257→      ref={ref}
   258→   258→      initial={{ opacity: 0, y: 40 }}
   259→   259→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   260→   260→      transition={{ duration: 0.6, ease: "easeOut", delay }}
   261→   261→      className={className}
   262→   262→    >
   263→   263→      {children}
   264→   264→    </motion.div>
   265→   265→  );
   266→   266→}

   272→   272→function Navigation() {

   274→   274→  const [mobileOpen, setMobileOpen] = useState(false);
   275→   275→

   279→   279→    return () => window.removeEventListener("scroll", onScroll);
   280→   280→  }, []);
   281→   281→
   282→   282→  return (
   283→   283→    <motion.header

   290→   290→          : "bg-transparent"

  1064→  1064→function ScrollToTop() {
  1065→  1065→  const [visible, setVisible] = useState(false);
  1066→  1066→
  1067→  1067→  useEffect(() => {
  1068→  1068→    const onScroll = () => setVisible(window.scrollY > 600);
  1069→  1069→    window.addEventListener("scroll", onScroll, { passive: true });
  1070→  1070→    return () => window.removeEventListener("scroll", onScroll);
  1071→  1071→  }, []);
  1072→  1072→
  1073→  1073→  return (
  1074→  1074→    <AnimatePresence>
  1075→  1075→      {visible && (
  1076→  1076→        <motion.button
  1077→  1077→          initial={{ opacity: 0, scale: 0.8 }}
  1078→  1078→          animate={{ opacity: 1, scale: 1 }}
  1079→  1079→          exit={{ opacity: 0, scale: 0.8 }}
  1080→  1080→          onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
  1081→  1081→          className="fixed bottom-6 right-6 z-50 w-11 h-11 rounded-full flex items-center justify-center shadow-lg hover:shadow-xl transition-shadow duration-300"
  1082→  1082→          style={{ background: "var(--pastel-sage)", color: "#fff" }}
  1083→  1083→          aria-label="Наверх"
  1084→  1084→        >
  1085→  1085→          <ArrowUp size={18} />
  1086→  1086→        </motion.button>
  1087→  1087→      )}
