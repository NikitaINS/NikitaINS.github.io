   241→/*  ANIMATED SECTION WRAPPER                                          */
   242→/* ------------------------------------------------------------------ */
   243→
   244→function FadeInSection({
   245→  children,
   246→  className = "",
   247→  delay = 0,
   248→}: {
   249→  children: React.ReactNode;
   250→  className?: string;
   251→  delay?: number;
   252→}) {
   253→  const ref = useRef<HTMLDivElement>(null);
   254→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   255→  return (
   256→    <motion.div
   257→      ref={ref}
   258→      initial={{ opacity: 0, y: 40 }}
   259→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   260→      transition={{ duration: 0.6, ease: "easeOut", delay }}
   261→      className={className}
   262→    >
   263→      {children}
   264→    </motion.div>
   265→  );
   266→}

   271→
   272→function Navigation() {
   273→  const [scrolled, setScrolled] = useState(false);
   274→  const [mobileOpen, setMobileOpen] = useState(false);
   275→
   276→  useEffect(() => {
   277→    const onScroll = () => setScrolled(window.scrollY > 40);
   278→    window.addEventListener("scroll", onScroll, { passive: true });
   279→    return () => window.removeEventListener("scroll", onScroll);
   280→  }, []);
   281→
   282→  return (
   283→    <motion.header
   284→      initial={{ y: -80 }}
   285→      animate={{ y: 0 }}
   286→      transition={{ duration: 0.5, ease: "easeOut" }}
   287→      className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
   288→        scrolled
   289→          ? "bg-white/80 backdrop-blur-md shadow-[0_2px_20px_rgba(0,0,0,0.06)]"
   290→          : "bg-transparent"
   291→      }`}
   292→    >
   293→      <nav className="mx-auto max-w-6xl px-6 h-16 flex items-center justify-between">
   294→        <a
   295→          href="#hero"
   296→          className="font-[family-name:var(--font-heading)] font-bold text-lg tracking-tight"
   297→          style={{ color: "var(--pastel-sage)" }}
   298→        >
   299→          Н. Иванов
   300→        </a>
   301→
   302→        {/* Desktop nav */}
   303→        <ul className="hidden md:flex items-center gap-8">
   304→          {NAV_ITEMS.map((item) => (
   305→            <li key={item.href}>
   306→              <a
   307→                href={item.href}
   308→                className="text-sm font-medium text-[var(--muted-foreground)] hover:text-[var(--foreground)] transition-colors duration-200"
   309→              >
   310→                {item.label}
   311→              </a>
   312→            </li>
   313→          ))}
   314→        </ul>
   315→
   316→        {/* Mobile toggle */}
   317→        <button
   318→          className="md:hidden p-2 rounded-lg hover:bg-[var(--secondary)] transition"
   319→          onClick={() => setMobileOpen(!mobileOpen)}
   320→          aria-label="Toggle menu"
   321→        >
   322→          {mobileOpen ? <X size={22} /> : <Menu size={22} />}
   323→        </button>
   324→      </nav>
   325→
   326→      {/* Mobile menu */}
   327→      <AnimatePresence>
   328→        {mobileOpen && (
   329→          <motion.div
   330→            initial={{ opacity: 0, height: 0 }}
   331→            animate={{ opacity: 1, height: "auto" }}
   332→            exit={{ opacity: 0, height: 0 }}
   333→            className="md:hidden bg-white/95 backdrop-blur-md border-b border-[var(--border)]"
   334→          >
   335→            <ul className="flex flex-col px-6 py-4 gap-3">
   336→              {NAV_ITEMS.map((item) => (
   337→                <li key={item.href}>
   338→                  <a
   339→                    href={item.href}
   340→                    onClick={() => setMobileOpen(false)}
   341→                    className="block py-2 text-sm font-medium text-[var(--muted-foreground)] hover:text-[var(--foreground)] transition-colors"
   342→                  >
   343→                    {item.label}
   344→                  </a>
   345→                </li>
   346→              ))}
   347→            </ul>
   348→          </motion.div>
   349→        )}
   350→      </AnimatePresence>
   351→    </motion.header>
   352→  );
   353→}

  1062→/* ------------------------------------------------------------------ */
  1063→
  1064→function ScrollToTop() {
  1065→  const [visible, setVisible] = useState(false);
  1066→
  1067→  useEffect(() => {
  1068→    const onScroll = () => setVisible(window.scrollY > 600);
  1069→    window.addEventListener("scroll", onScroll, { passive: true });
  1070→    return () => window.removeEventListener("scroll", onScroll);
  1071→  }, []);
  1072→
  1073→  return (
  1074→    <AnimatePresence>
  1075→      {visible && (
  1076→        <motion.button
  1077→          initial={{ opacity: 0, scale: 0.8 }}
  1078→          animate={{ opacity: 1, scale: 1 }}
  1079→          exit={{ opacity: 0, scale: 0.8 }}
  1080→          onClick={() => window.scrollTo({ top: 0, behavior: "smooth" })}
  1081→          className="fixed bottom-6 right-6 z-50 w-11 h-11 rounded-full flex items-center justify-center shadow-lg hover:shadow-xl transition-shadow duration-300"
  1082→          style={{ background: "var(--pastel-sage)", color: "#fff" }}
  1083→          aria-label="Наверх"
  1084→        >
  1085→          <ArrowUp size={18} />
  1086→        </motion.button>
  1087→      )}
  1088→    </AnimatePresence>
  1089→  );
  1090→}
